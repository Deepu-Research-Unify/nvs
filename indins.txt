  <script>
    let qrInstance = null;

    function syncCard() {
      const taluka = document.getElementById("inTaluka").value.trim() || "—";
      const district = document.getElementById("inDistrict").value.trim() || "—";
      const state = document.getElementById("inState").value.trim() || "—";
      const region = document.getElementById("inRegion").value.trim() || "—";

      document.getElementById("dispCommonLocation").innerText =
        `TALUKA: ${taluka} • DIST: ${district} • STATE: ${state} • REGION: ${region}`;

      const name = document.getElementById("inName").value.trim() || "Student Name";
      const gr = document.getElementById("inGR").value.trim() || "000";
      const batch = document.getElementById("inBatch").value.trim() || "2026-27";

      document.getElementById("dispBatch").innerText = batch;
      document.getElementById("dispName").innerText = name;
      document.getElementById("dispFather").innerText = document.getElementById("inFather").value.trim() || "—";
      document.getElementById("dispMother").innerText = document.getElementById("inMother").value.trim() || "—";
      document.getElementById("dispDOB").innerText = document.getElementById("inDOB").value || "—";
      document.getElementById("dispHouse").innerText = document.getElementById("inHouse").value.trim() || "—";
      document.getElementById("dispGR").innerText = gr;
      document.getElementById("dispMobile").innerText = document.getElementById("inMobile").value.trim() || "—";
      document.getElementById("dispSign").innerText = document.getElementById("inSign").value.trim() || "Principal";
      document.getElementById("dispCardCode").innerText = gr;

      updateQR(`PM SHRI JNV | ${name} | GR: ${gr} | Dist: ${district} (${state})`);
    }

    function updateQR(text) {
      const qrBox = document.getElementById("miniQr");
      qrBox.innerHTML = "";
      qrInstance = new QRCode(qrBox, {
        text: text,
        width: 64,
        height: 64,
        colorDark: "#06255c",
        colorLight: "#ffffff",
        correctLevel: QRCode.CorrectLevel.M
      });
    }

    document.getElementById("inPhoto").addEventListener("change", function (e) {
      const file = e.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = function (ev) {
        document.getElementById("studentPhoto").src = ev.target.result;
      };
      reader.readAsDataURL(file);
    });

    function adjustPhoto() {
      const z = document.getElementById("photoZoom").value;
      const x = document.getElementById("photoX").value;
      const y = document.getElementById("photoY").value;
      document.getElementById("zoomVal").innerText = `${parseFloat(z).toFixed(2)}x`;
      document.getElementById("studentPhoto").style.transform = `scale(${z}) translate(${x}px, ${y}px)`;
    }

    function printCleanCard() {
      syncCard();
      window.print();
    }

    function exportCardPNG() {
      syncCard();
      const card = document.getElementById("idCard");
      html2canvas(card, {
        scale: 3,
        useCORS: true,
        backgroundColor: null
      }).then((canvas) => {
        const dataUrl = canvas.toDataURL("image/png");
